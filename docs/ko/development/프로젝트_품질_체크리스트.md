```python
# scripts/monitor_quality.py
"""프로젝트 품질 지표 모니터링"""

import subprocess
import json
from datetime import datetime

def run_quality_checks():
    """품질 검사 실행 및 결과 수집"""
    
    results = {
        "timestamp": datetime.now().isoformat(),
        "lint_errors": 0,
        "test_coverage": 0.0,
        "test_failures": 0,
        "complexity_warnings": 0
    }
    
    # Ruff 검사
    try:
        result = subprocess.run(
            ["ruff", "check", "src/", "--output-format=json"],
            capture_output=True, text=True
        )
        if result.stdout:
            lint_data = json.loads(result.stdout)
            results["lint_errors"] = len(lint_data)
    except Exception:
        pass
    
    # 테스트 커버리지
    try:
        result = subprocess.run(
            ["pytest", "--cov=src", "--cov-report=json"],
            capture_output=True
        )
        with open("coverage.json", "r") as f:
            cov_data = json.load(f)
            results["test_coverage"] = cov_data["totals"]["percent_covered"]
    except Exception:
        pass
    
    return results

if __name__ == "__main__":
    quality_metrics = run_quality_checks()
    print(json.dumps(quality_metrics, indent=2))
```
