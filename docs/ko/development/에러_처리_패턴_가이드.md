```python
@staticmethod
def _validate_and_fix_date_range(start_date: str) -> str:
    """날짜 범위 검증 및 미래 날짜 보정"""
    try:
        # 현재 날짜
        current_date = datetime.now()
        current_year_month = current_date.strftime("%Y-%m")
        
        # 입력 날짜 파싱 및 검증
        if not start_date or len(start_date) != 7:  # YYYY-MM 형식
            return current_year_month
        
        start_year, start_month = map(int, start_date.split("-"))
        start_datetime = datetime(start_year, start_month, 1)
        
        # 미래 날짜 검증
        if start_datetime > current_date:
            _LOGGER.warning(f"Future date detected: {start_date}, using current: {current_year_month}")
            return current_year_month
        
        # 너무 과거 날짜 검증 (5년 이전)
        five_years_ago = current_date - timedelta(days=365 * 5)
        if start_datetime < five_years_ago:
            safe_start = five_years_ago.strftime("%Y-%m")
            _LOGGER.warning(f"Too old date: {start_date}, using: {safe_start}")
            return safe_start
        
        return start_date
        
    except Exception as e:
        _LOGGER.error(f"Date validation failed: {e}")
        return datetime.now().strftime("%Y-%m")
```

```python
from functools import wraps
import time

def retry_on_failure(
    max_retries: int = 3,
    delay: float = 1.0, 
    backoff_factor: float = 2.0,
    exceptions: tuple = (Exception,)
):
    """API 호출 실패 시 재시도 데코레이터"""
    
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            current_delay = delay
            last_exception = None
            
            for attempt in range(max_retries + 1):
                try:
                    return func(*args, **kwargs)
                    
                except exceptions as e:
                    last_exception = e
                    
                    if attempt == max_retries:
                        _LOGGER.error(
                            f"Function {func.__name__} failed after {max_retries} retries: {e}"
                        )
                        raise
                    
                    _LOGGER.warning(
                        f"Function {func.__name__} attempt {attempt + 1} failed: {e}, "
                        f"retrying in {current_delay}s..."
                    )
                    
                    time.sleep(current_delay)
                    current_delay *= backoff_factor
            
            if last_exception:
                raise last_exception
                
        return wrapper
    return decorator

# 사용 예시
@retry_on_failure(max_retries=3, delay=2.0)
def _call_bigquery_api(self, query: str):
    """BigQuery API 호출 with 재시도"""
    return self.bigquery_connector.read_df_from_bigquery(query)
```

```python
class GracefulErrorHandler:
    """작업 실패 시 우아한 복구를 위한 에러 핸들러"""
    
    def __init__(self, operation_name: str):
        self.operation_name = operation_name
        self.error_count = 0
        self.last_error = None
    
    def handle_error(self, error: Exception, context: str = "") -> bool:
        """에러를 처리하고 계속 진행할지 결정"""
        
        self.error_count += 1
        self.last_error = error
        error_info = f"{context}: {error}" if context else str(error)
        
        _LOGGER.warning(
            f"[GracefulErrorHandler] {self.operation_name} - "
            f"Error #{self.error_count}: {error_info}"
        )
        
        # 에러 타입별 처리 전략
        error_str = str(error).lower()
        
        if "permission" in error_str or "access" in error_str:
            _LOGGER.info(f"Access/permission error detected, continuing with partial results")
            return True  # 권한 에러는 계속 진행
            
        elif "timeout" in error_str or "connection" in error_str:
            _LOGGER.info(f"Network error detected, continuing with partial results")
            return True  # 네트워크 에러는 계속 진행
            
        elif self.error_count < 5:  # 최대 5개 에러까지 허용
            return True
            
        else:
            _LOGGER.error(f"Too many errors ({self.error_count}), stopping operation")
            return False
    
    def get_summary(self) -> dict:
        """에러 처리 요약 정보 반환"""
        return {
            "operation_name": self.operation_name,
            "error_count": self.error_count,
            "last_error": str(self.last_error) if self.last_error else None,
            "status": "completed_with_errors" if self.error_count > 0 else "success"
        }

# 사용 예시
def _process_multiple_files(self, files: list):
    """여러 파일 처리 with 우아한 에러 처리"""
    
    error_handler = GracefulErrorHandler("file_processing")
    successful_files = 0
    
    for file_info in files:
        try:
            result = self._process_single_file(file_info)
            successful_files += 1
            yield result
            
        except Exception as e:
            if not error_handler.handle_error(e, f"file: {file_info['name']}"):
                break  # 너무 많은 에러 발생 시 중단
            continue  # 개별 파일 실패는 무시하고 계속
    
    # 처리 요약 로깅
    summary = error_handler.get_summary()
    _LOGGER.info(f"File processing completed: {successful_files} successful, "
                f"{summary['error_count']} errors")
```

```python
def parse_stream(self, stream, field_mapper, **kwargs):
    """스트림 파싱 with 레코드별 에러 처리"""
    
    processed_count = 0
    error_count = 0
    
    try:
        for batch in self._read_batches(stream):
            batch_records = []
            
            for row_index, row in enumerate(batch):
                try:
                    # 개별 레코드 처리
                    row_dict = self._convert_row_to_dict(row)
                    row_dict = self._clean_nan_values(row_dict)
                    
                    # 필드 매핑
                    mapped_record = field_mapper.map_record(row_dict)
                    
                    # 필수 필드 검증
                    self._validate_required_fields(mapped_record)
                    
                    batch_records.append(mapped_record)
                    processed_count += 1
                    
                except Exception as e:
                    error_count += 1
                    _LOGGER.warning(
                        f"Failed to process row {processed_count + error_count}: {e}"
                    )
                    
                    # 에러율이 너무 높으면 중단
                    if error_count > processed_count * 0.1:  # 10% 에러율 임계값
                        _LOGGER.error(f"Error rate too high: {error_count}/{processed_count + error_count}")
                        raise Exception("Too many parsing errors")
                    
                    continue  # 개별 레코드 실패는 건너뛰기
            
            # 배치 결과 반환
            if batch_records:
                yield {"results": batch_records}
                
    except Exception as e:
        _LOGGER.error(f"Stream parsing failed: {e}")
        raise
    
    finally:
        _LOGGER.info(f"Parsing completed: {processed_count} processed, {error_count} errors")
```

```python
def map_record(self, source_data: dict) -> dict:
    """레코드 매핑 with 필드별 에러 처리"""
    
    try:
        # 핵심 필드 매핑
        mapped_data = {}
        
        # 필수 필드들을 안전하게 매핑
        essential_fields = ["cost", "billed_date", "provider"]
        for field in essential_fields:
            try:
                mapped_data[field] = self._map_field_safely(field, source_data)
            except Exception as e:
                _LOGGER.error(f"Essential field mapping failed: {field}, error: {e}")
                raise ERROR_INVALID_ARGUMENT(key=f"field_mapper_error: {field}")
        
        # 선택적 필드들을 안전하게 매핑
        optional_fields = ["usage_quantity", "region_code", "product", "usage_type"]
        for field in optional_fields:
            try:
                mapped_data[field] = self._map_field_safely(field, source_data)
            except Exception as e:
                _LOGGER.warning(f"Optional field mapping failed: {field}, using default")
                mapped_data[field] = self._get_field_default(field)
        
        # SpaceONE data 필드 생성 (필수)
        try:
            listed_price = self._get_listed_price_from_source(source_data)
            mapped_data["data"] = self._create_spaceone_billing_data(
                source_data, mapped_data, listed_price
            )
        except Exception as e:
            _LOGGER.error(f"SpaceONE data field creation failed: {e}")
            mapped_data["data"] = {}  # 빈 딕셔너리로 보장
        
        return mapped_data
        
    except Exception as e:
        _LOGGER.error(f"Record mapping failed: {e}")
        raise ERROR_INVALID_ARGUMENT(key=f"field_mapper_error: {str(e)}")

def _map_field_safely(self, field_name: str, source_data: dict):
    """필드를 안전하게 매핑"""
    try:
        if field_name in self.compiled_mappings:
            return self.compiled_mappings[field_name](source_data)
        else:
            return self._get_nested_value(source_data, field_name, self._get_field_default(field_name))
    except Exception as e:
        _LOGGER.warning(f"Field mapping failed for {field_name}: {e}")
        return self._get_field_default(field_name)

def _get_field_default(self, field_name: str):
    """필드별 기본값 반환"""
    defaults = {
        "cost": 0,
        "usage_quantity": 0, 
        "provider": "unknown",
        "region_code": "global",
        "product": "Unknown",
        "usage_type": "",
        "resource": "",
        "billed_date": datetime.now().strftime("%Y-%m-%d"),
        "tags": {},
        "additional_info": {}
    }
    return defaults.get(field_name, "")
```

```python
def _sanitize_for_serialization(self, data: dict) -> dict:
    """JSON 직렬화 안전성 보장"""
    
    import pandas as pd
    import numpy as np
    from datetime import datetime, date
    from decimal import Decimal
    
    def convert_value_safely(value):
        """개별 값을 안전하게 변환"""
        try:
            # None 체크 우선
            if value is None:
                return None
            
            # Pandas NA 안전 체크
            try:
                if pd.isna(value):
                    return None
            except (TypeError, ValueError):
                pass
            
            # 타입별 안전한 변환
            if isinstance(value, (pd.Timestamp, pd.Timedelta)):
                return str(value)
            elif isinstance(value, (np.integer, np.floating)):
                return value.item()
            elif isinstance(value, np.ndarray):
                return [convert_value_safely(item) for item in value]
            elif isinstance(value, (datetime, date)):
                return value.isoformat()
            elif isinstance(value, Decimal):
                return float(value)
            elif isinstance(value, dict):
                return {k: convert_value_safely(v) for k, v in value.items()}
            elif isinstance(value, (list, tuple)):
                return [convert_value_safely(item) for item in value]
            else:
                return value
                
        except Exception as e:
            _LOGGER.warning(f"Value conversion failed: {value}, error: {e}")
            return str(value) if value is not None else None
    
    # 전체 데이터 안전 변환
    sanitized_data = {}
    
    for key, value in data.items():
        try:
            sanitized_value = convert_value_safely(value)
            
            # SpaceONE 필수 필드 특별 처리
            if key == "data" and not isinstance(sanitized_value, dict):
                _LOGGER.warning(f"data field is not dict, forcing empty dict")
                sanitized_data[key] = {}
            else:
                sanitized_data[key] = sanitized_value
                
        except Exception as e:
            _LOGGER.warning(f"Field sanitization failed for {key}: {e}")
            
            # 필수 필드별 안전 처리
            if key == "data":
                sanitized_data[key] = {}
            elif key == "billed_date":
                sanitized_data[key] = datetime.now().strftime("%Y-%m-%d")
            elif key in ["tags", "additional_info"]:
                sanitized_data[key] = {}
            else:
                sanitized_data[key] = str(value) if value is not None else ""
    
    return sanitized_data
```

```python
class ErrorStatistics:
    """에러 통계 수집 클래스"""
    
    def __init__(self):
        self.error_counts = {}
        self.total_operations = 0
        self.start_time = datetime.now()
    
    def record_error(self, error_type: str, context: str = ""):
        """에러 발생 기록"""
        key = f"{error_type}:{context}" if context else error_type
        self.error_counts[key] = self.error_counts.get(key, 0) + 1
    
    def record_success(self):
        """성공 작업 기록"""
        self.total_operations += 1
    
    def get_summary(self) -> dict:
        """에러 통계 요약"""
        total_errors = sum(self.error_counts.values())
        duration = datetime.now() - self.start_time
        
        return {
            "total_operations": self.total_operations,
            "total_errors": total_errors,
            "error_rate": total_errors / max(self.total_operations, 1),
            "duration_seconds": duration.total_seconds(),
            "error_breakdown": self.error_counts
        }
```

```python
def debug_error_context(func):
    """에러 발생 시 컨텍스트 정보 로깅 데코레이터"""
    
    @wraps(func)
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            # 컨텍스트 정보 수집
            context_info = {
                "function": func.__name__,
                "args_count": len(args),
                "kwargs_keys": list(kwargs.keys()),
                "error_type": type(e).__name__,
                "error_message": str(e)
            }
            
            _LOGGER.error(f"Error context: {context_info}", exc_info=True)
            raise
    
    return wrapper
```
