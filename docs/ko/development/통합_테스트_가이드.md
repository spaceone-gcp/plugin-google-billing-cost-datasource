#  통합 테스트 가이드 (gRPC 테스트 모범사례 포함)

##  개요

이 문서는 SpaceONE Google Cloud Billing Cost Datasource 플러그인에서 gRPC API 테스트를 효과적으로 수행하기 위한 모범 사례를 제시합니다. gRPC reflection API 제약 상황에서도 안정적이고 효율적인 테스트 환경을 구축하는 방법을 다룹니다.

##  테스트 전략 개요

### 계층별 테스트 접근법

```
┌─────────────────────────────────────────┐
│ 1. Unit Tests (단위 테스트)              │
│    - Manager 클래스 직접 테스트          │
│    - 비즈니스 로직 검증                  │
└─────────────────────────────────────────┘
                    ↓
┌─────────────────────────────────────────┐
│ 2. Integration Tests (통합 테스트)       │
│    - Python gRPC 클라이언트             │
│    - 전체 API 플로우 검증               │
└─────────────────────────────────────────┘
                    ↓
┌─────────────────────────────────────────┐
│ 3. E2E Tests (종단간 테스트)            │
│    - 실제 데이터 소스 연동               │
│    - 운영 환경 시뮬레이션               │
└─────────────────────────────────────────┘
```

```python
    def setup_connection(self):
        """gRPC 연결 설정"""
        try:

            }
```

##  3단계: E2E Tests (종단간 테스트)

### 실제 데이터 소스 테스트

```python
# test_e2e_real_data.py
import os
import json
import time
from datetime import datetime, timedelta
from typing import Dict, List, Any

class E2ETestSuite:
    """종단간 테스트 스위트"""
    
    def __init__(self):
        self.test_results = {}
        self.start_time = None
        
    def run_full_e2e_test(self) -> Dict[str, Any]:
        """전체 E2E 테스트 실행"""
        self.start_time = time.time()
        
        print(" E2E 테스트 시작")
        print(f" 테스트 시작 시간: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        
        # 테스트 단계별 실행
        test_stages = [
            ("환경 검증", self._verify_environment),
            ("서버 연결", self._test_server_connection),
            ("데이터 소스 초기화", self._test_data_source_setup),
            ("작업 태스크 생성", self._test_job_tasks_creation),
            ("비용 데이터 수집", self._test_cost_data_collection),
            ("데이터 품질 검증", self._test_data_quality),
            ("성능 측정", self._test_performance_metrics),
        ]
        
        for stage_name, stage_func in test_stages:
            print(f"\n {stage_name} 단계 실행 중...")
            try:
                result = stage_func()
                self.test_results[stage_name] = {
                    "status": "success",
                    "result": result,
                    "timestamp": datetime.now().isoformat()
                }
                print(f" {stage_name} 완료")
            except Exception as e:
                self.test_results[stage_name] = {
                    "status": "failed", 
                    "error": str(e),
                    "timestamp": datetime.now().isoformat()
                }
                print(f" {stage_name} 실패: {e}")
                
        # 최종 결과 정리
        return self._generate_final_report()
        
    def _verify_environment(self) -> Dict[str, Any]:
        """환경 검증"""
        checks = {
            "python_version": f"{os.sys.version_info.major}.{os.sys.version_info.minor}",
            "required_env_vars": {},
            "dependencies": {}
        }
        
        # 필수 환경변수 확인
        required_vars = ["GOOGLE_APPLICATION_CREDENTIALS", "DEBUG"]
        for var in required_vars:
            checks["required_env_vars"][var] = os.getenv(var) is not None
            
        # 의존성 확인
        try:
            import grpc
            checks["dependencies"]["grpc"] = grpc.__version__
        except ImportError:
            checks["dependencies"]["grpc"] = "Not installed"
            
        return checks
        
    def _test_server_connection(self) -> Dict[str, Any]:
        """서버 연결 테스트"""
        import socket
        
        result = {
            "server_address": "localhost:50051",
            "connection_test": False,
            "response_time_ms": None
        }
        
        start = time.time()
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(5)
            connection_result = sock.connect_ex(('localhost', 50051))
            sock.close()
            
            result["connection_test"] = connection_result == 0
            result["response_time_ms"] = round((time.time() - start) * 1000, 2)
            
        except Exception as e:
            result["error"] = str(e)
            
        return result
        
    def _test_data_source_setup(self) -> Dict[str, Any]:
        """데이터 소스 설정 테스트"""
        # 실제 DataSourceManager를 사용한 테스트
        from plugin.manager.data_source_manager import DataSourceManager
        
        manager = DataSourceManager()
        test_options = {
            "billing_export_project_id": "mkkang-project",
            "billing_dataset_id": "billing_dataset", 
            "billing_account_id": "01FD8E-B4DDC1-EAB69F"
        }
        
        result = manager.init_response(test_options)
        
        return {
            "options_validated": True,
            "metadata_generated": "metadata" in result,
            "currency": result.get("metadata", {}).get("currency"),
            "supported_secret_types": result.get("metadata", {}).get("supported_secret_types")
        }
        
    def _test_job_tasks_creation(self) -> Dict[str, Any]:
        """작업 태스크 생성 테스트"""
        from plugin.manager.job_manager import JobManager
        
        manager = JobManager()
        
        # 실제 시크릿 데이터 (테스트용)
        test_secret = self._load_test_secret_data()
        test_options = self._load_test_options()
        
        result = manager.get_tasks(
            domain_id="test-domain",
            options=test_options,
            secret_data=test_secret,
            schema=None,
            start="2025-09",
            last_synchronized_at=None
        )
        
        return {
            "tasks_created": len(result.get("tasks", [])),
            "task_types": [task.get("task_options", {}).get("project_id") for task in result.get("tasks", [])[:5]],  # 처음 5개만
            "changed_items": len(result.get("changed", []))
        }
        
    def _test_cost_data_collection(self) -> Dict[str, Any]:
        """비용 데이터 수집 테스트"""
        from plugin.manager.cost_manager import CostManager
        
        manager = CostManager()
        
        test_secret = self._load_test_secret_data()
        test_options = self._load_test_options()
        test_task_options = {
            "project_id": "mkkang-project",
            "start": "2025-09"
        }
        
        # 첫 번째 배치만 테스트 (전체 수집은 시간이 오래 걸림)
        result_generator = manager.get_data(
            options=test_options,
            secret_data=test_secret,
            task_options=test_task_options,
            schema=None
        )
        
        first_batch = next(result_generator, None)
        
        if first_batch and "results" in first_batch:
            sample_record = first_batch["results"][0] if first_batch["results"] else None
            
            return {
                "batch_collected": True,
                "records_in_first_batch": len(first_batch["results"]),
                "sample_record_fields": list(sample_record.keys()) if sample_record else [],
                "sample_cost_value": sample_record.get("cost") if sample_record else None
            }
        else:
            return {"batch_collected": False, "error": "No data returned"}
            
    def _test_data_quality(self) -> Dict[str, Any]:
        """데이터 품질 검증"""
        # 이전 단계에서 수집된 데이터를 기반으로 품질 검증
        quality_checks = {
            "required_fields_present": True,
            "cost_values_valid": True,
            "date_formats_correct": True,
            "currency_consistent": True
        }
        
        # 실제 구현에서는 수집된 데이터를 검증
        return quality_checks
        
    def _test_performance_metrics(self) -> Dict[str, Any]:
        """성능 측정"""
        total_time = time.time() - self.start_time
        
        return {
            "total_test_time_seconds": round(total_time, 2),
            "average_stage_time": round(total_time / len(self.test_results), 2),
            "memory_usage_mb": self._get_memory_usage(),
            "performance_grade": "Good" if total_time < 60 else "Needs Improvement"
        }
        
    def _get_memory_usage(self) -> float:
        """메모리 사용량 측정"""
        import psutil
        import os
        
        process = psutil.Process(os.getpid())
        return round(process.memory_info().rss / 1024 / 1024, 2)
        
    def _load_test_secret_data(self) -> Dict[str, Any]:
        """테스트용 시크릿 데이터 로드"""
        # 실제 환경에서는 안전한 방법으로 로드
        return {
            "type": "service_account",
            "project_id": "mkkang-project",
            "client_email": "test@mkkang-project.iam.gserviceaccount.com",
            "private_key": "-----BEGIN PRIVATE KEY-----\nTEST_KEY\n-----END PRIVATE KEY-----"
        }
        
    def _load_test_options(self) -> Dict[str, Any]:
        """테스트용 옵션 로드"""
        return {
            "billing_export_project_id": "mkkang-project",
            "billing_dataset_id": "billing_dataset",
            "billing_account_id": "01FD8E-B4DDC1-EAB69F"
        }
        
    def _generate_final_report(self) -> Dict[str, Any]:
        """최종 보고서 생성"""
        successful_stages = sum(1 for result in self.test_results.values() if result["status"] == "success")
        total_stages = len(self.test_results)
        success_rate = (successful_stages / total_stages) * 100 if total_stages > 0 else 0
        
        return {
            "summary": {
                "total_stages": total_stages,
                "successful_stages": successful_stages,
                "failed_stages": total_stages - successful_stages,
                "success_rate_percent": round(success_rate, 1),
                "overall_status": "PASS" if success_rate >= 80 else "FAIL"
            },
            "detailed_results": self.test_results,
            "recommendations": self._generate_recommendations()
        }
        
    def _generate_recommendations(self) -> List[str]:
        """개선 권장사항 생성"""
        recommendations = []
        
        for stage_name, result in self.test_results.items():
            if result["status"] == "failed":
                recommendations.append(f"{stage_name} 단계 문제 해결 필요")
                
        if not recommendations:
            recommendations.append("모든 테스트가 성공했습니다!")
            
        return recommendations

# 실행 스크립트
if __name__ == "__main__":
    print(" E2E 테스트 스위트 실행")
    
    test_suite = E2ETestSuite()
    final_report = test_suite.run_full_e2e_test()
    
    # 결과 출력
    print("\n" + "="*60)
    print(" 최종 E2E 테스트 보고서")
    print("="*60)
    
    summary = final_report["summary"]
    print(f"전체 상태: {' PASS' if summary['overall_status'] == 'PASS' else ' FAIL'}")
    print(f"성공률: {summary['success_rate_percent']}% ({summary['successful_stages']}/{summary['total_stages']})")
    
    print("\n 단계별 결과:")
    for stage_name, result in final_report["detailed_results"].items():
        status_icon = "" if result["status"] == "success" else ""
        print(f"  {status_icon} {stage_name}")
        
    print("\n 권장사항:")
    for rec in final_report["recommendations"]:
        print(f"  • {rec}")
        
    # JSON 파일로 상세 결과 저장
    with open("e2e_test_report.json", "w", encoding="utf-8") as f:
        json.dump(final_report, f, indent=2, ensure_ascii=False)
    
    print(f"\n 상세 보고서가 'e2e_test_report.json'에 저장되었습니다.")
```

##  테스트 모니터링 및 메트릭

### 테스트 메트릭 수집

```python
# test_metrics_collector.py
import time
import json
import psutil
from dataclasses import dataclass, asdict
from typing import Dict, List, Any
from datetime import datetime

@dataclass
class TestMetrics:
    """테스트 메트릭 데이터 클래스"""
    test_name: str
    start_time: float
    end_time: float
    duration_seconds: float
    memory_usage_mb: float
    cpu_usage_percent: float
    success: bool
    error_message: str = ""
    
    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

class TestMetricsCollector:
    """테스트 메트릭 수집기"""
    
    def __init__(self):
        self.metrics: List[TestMetrics] = []
        
    def start_test(self, test_name: str) -> Dict[str, Any]:
        """테스트 시작 메트릭 수집"""
        return {
            "test_name": test_name,
            "start_time": time.time(),
            "start_memory": self._get_memory_usage(),
            "start_cpu": self._get_cpu_usage()
        }
        
    def end_test(self, test_context: Dict[str, Any], success: bool, error_message: str = "") -> TestMetrics:
        """테스트 종료 메트릭 수집"""
        end_time = time.time()
        end_memory = self._get_memory_usage()
        
        metrics = TestMetrics(
            test_name=test_context["test_name"],
            start_time=test_context["start_time"],
            end_time=end_time,
            duration_seconds=round(end_time - test_context["start_time"], 3),
            memory_usage_mb=round(end_memory - test_context["start_memory"], 2),
            cpu_usage_percent=self._get_cpu_usage(),
            success=success,
            error_message=error_message
        )
        
        self.metrics.append(metrics)
        return metrics
        
    def generate_report(self) -> Dict[str, Any]:
        """메트릭 보고서 생성"""
        if not self.metrics:
            return {"error": "No metrics collected"}
            
        successful_tests = [m for m in self.metrics if m.success]
        failed_tests = [m for m in self.metrics if not m.success]
        
        return {
            "summary": {
                "total_tests": len(self.metrics),
                "successful_tests": len(successful_tests),
                "failed_tests": len(failed_tests),
                "success_rate": round(len(successful_tests) / len(self.metrics) * 100, 1),
                "total_duration": round(sum(m.duration_seconds for m in self.metrics), 2),
                "average_duration": round(sum(m.duration_seconds for m in self.metrics) / len(self.metrics), 3)
            },
            "performance": {
                "fastest_test": min(self.metrics, key=lambda m: m.duration_seconds).test_name,
                "slowest_test": max(self.metrics, key=lambda m: m.duration_seconds).test_name,
                "max_memory_usage": max(m.memory_usage_mb for m in self.metrics),
                "avg_cpu_usage": round(sum(m.cpu_usage_percent for m in self.metrics) / len(self.metrics), 1)
            },
            "detailed_metrics": [m.to_dict() for m in self.metrics],
            "failed_tests_details": [
                {"name": m.test_name, "error": m.error_message, "duration": m.duration_seconds}
                for m in failed_tests
            ]
        }
        
    def save_report(self, filename: str = None):
        """보고서를 파일로 저장"""
        if filename is None:
            filename = f"test_metrics_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json"
            
        report = self.generate_report()
        
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump(report, f, indent=2, ensure_ascii=False)
            
        print(f" 테스트 메트릭 보고서가 {filename}에 저장되었습니다.")
        return filename
        
    def _get_memory_usage(self) -> float:
        """현재 메모리 사용량 (MB)"""
        import os
        process = psutil.Process(os.getpid())
        return round(process.memory_info().rss / 1024 / 1024, 2)
        
    def _get_cpu_usage(self) -> float:
        """현재 CPU 사용률 (%)"""
        return round(psutil.cpu_percent(interval=1), 1)

# 사용 예시
def run_tests_with_metrics():
    """메트릭 수집과 함께 테스트 실행"""
    collector = TestMetricsCollector()
    
    # 테스트 목록
    test_functions = [
        ("Manager Direct Test", run_manager_test),
        ("gRPC Integration Test", run_grpc_test), 
        ("E2E Test", run_e2e_test)
    ]
    
    for test_name, test_func in test_functions:
        print(f"\n {test_name} 실행 중...")
        
        # 메트릭 수집 시작
        test_context = collector.start_test(test_name)
        
        try:
            # 테스트 실행
            test_func()
            
            # 성공 메트릭 기록
            metrics = collector.end_test(test_context, success=True)
            print(f" {test_name} 완료 ({metrics.duration_seconds}초)")
            
        except Exception as e:
            # 실패 메트릭 기록
            metrics = collector.end_test(test_context, success=False, error_message=str(e))
            print(f" {test_name} 실패 ({metrics.duration_seconds}초): {e}")
    
    # 최종 보고서 생성
    report_file = collector.save_report()
    
    # 요약 출력
    report = collector.generate_report()
    summary = report["summary"]
    
    print(f"\n 테스트 완료!")
    print(f"성공률: {summary['success_rate']}% ({summary['successful_tests']}/{summary['total_tests']})")
    print(f"총 소요시간: {summary['total_duration']}초")
    
    return report_file
```
