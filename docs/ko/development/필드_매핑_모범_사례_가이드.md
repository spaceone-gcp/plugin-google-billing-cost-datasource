def _get_nested_value(self, data: dict, path: str, default_value: Any = "") -> Any:
    """중첩 객체 경로를 안전하게 처리"""
    
    if not path:
        return default_value
    
    try:
        if "." in path:
            # 점(.)으로 구분된 경로 처리
            keys = path.split(".")
            current_value = data
            
            for key in keys:
                if isinstance(current_value, dict):
                    current_value = current_value.get(key)
                    if current_value is None:
                        return default_value
                elif isinstance(current_value, str):
                    # JSON 문자열인 경우 파싱 시도
                    try:
                        import json
                        parsed_value = json.loads(current_value)
                        if isinstance(parsed_value, dict):
                            current_value = parsed_value.get(key)
                        else:
                            return default_value
                    except (json.JSONDecodeError, ValueError):
                        return default_value
                else:
                    return default_value
            
            return current_value if current_value is not None else default_value
        else:
            # 단순 필드명
            return data.get(path, default_value)
            
    except Exception as e:
        _LOGGER.warning(f"Failed to get nested value for path '{path}': {e}")
        return default_value
```

def _format_date(self, value: Any) -> str:
    """다양한 날짜 형식을 YYYY-MM-DD로 통일"""
    
    # 빈 값 처리
    if not value or str(value).strip() == "" or str(value) == "today":
        return datetime.now().strftime("%Y-%m-%d")
    
    try:
        # pandas.Timestamp 처리
        if hasattr(value, "to_pydatetime"):
            return value.to_pydatetime().strftime("%Y-%m-%d")
        elif isinstance(value, datetime):
            return value.strftime("%Y-%m-%d")
        elif isinstance(value, str):
            # 다양한 날짜 형식 지원
            date_formats = [
                "%Y-%m-%d",
                "%Y-%m-%d %H:%M:%S",
                "%Y-%m-%dT%H:%M:%S",
                "%Y-%m-%dT%H:%M:%SZ",
                "%Y-%m-%d %H:%M:%S UTC",           # Google Cloud 형식
                "%Y-%m-%d %H:%M:%S.%f UTC",       # 마이크로초 포함
                "%Y/%m/%d", "%m/%d/%Y", "%d/%m/%Y"
            ]
            
            for fmt in date_formats:
                try:
                    parsed_date = datetime.strptime(value, fmt)
                    return parsed_date.strftime("%Y-%m-%d")
                except ValueError:
                    continue
    
    except Exception as e:
        _LOGGER.warning(f"Date format failed: {value}, error: {e}")
    
    # 모든 시도 실패 시 현재 날짜
    return datetime.now().strftime("%Y-%m-%d")
```

### 2. JSON 파싱 (json_parse)

```python
def _apply_json_parse_transform(self, value: Any) -> dict:
    """안전한 JSON 파싱 with fallback"""
    
    if not value:
        return {}
    
    # 이미 딕셔너리인 경우
    if isinstance(value, dict):
        return value
    
    json_str = str(value).strip()
    if not json_str or json_str.lower() in ("none", "null", ""):
        return {}
    
    try:
        # 표준 JSON 파싱
        import json
        parsed = json.loads(json_str)
        return parsed if isinstance(parsed, dict) else {}
        
    except (json.JSONDecodeError, ValueError):
        # JSON 파싱 실패 시 대안 처리
        return self._parse_json_fallback(json_str)

def _parse_json_fallback(self, json_str: str) -> dict:
    """JSON 파싱 실패 시 key=value 형태 처리"""
    if "=" in json_str:
        try:
            pairs = {}
            for pair in json_str.split(","):
                if "=" in pair:
                    k, v = pair.split("=", 1)
                    pairs[k.strip()] = v.strip()
            return pairs
        except Exception:
            pass
    return {}
```

def _map_tags_field(self, source_data: dict) -> dict:
    """Google Cloud labels 배열을 딕셔너리로 변환"""
    
    tags_value = self._map_field("tags", source_data, {})
    
    # 이미 딕셔너리인 경우
    if isinstance(tags_value, dict):
        return tags_value
    
    # Google Cloud labels 배열 형태 처리
    if isinstance(tags_value, list):
        result_dict = {}
        for item in tags_value:
            if isinstance(item, dict) and "key" in item and "value" in item:
                result_dict[item["key"]] = item["value"]
        return result_dict
    
    # 문자열인 경우 JSON 파싱 시도
    if isinstance(tags_value, str) and tags_value.strip():
        try:
            import json
            parsed_tags = json.loads(tags_value)
            
            if isinstance(parsed_tags, dict):
                return parsed_tags
            elif isinstance(parsed_tags, list):
                # labels 배열 형태 처리
                return self._process_labels_array(parsed_tags)
                
        except (json.JSONDecodeError, ValueError):
            # key=value 형태 파싱 시도
            return self._parse_key_value_pairs(tags_value)
    
    return {}

### 1. Decimal 기반 정확한 비용 계산

```python
def _process_numeric_field(self, value: Any) -> Any:
    """연산 시에만 Decimal 사용, 원본 타입 유지"""
    
    if value is None:
        return 0
    
    # 원본 타입 저장
    original_type = type(value)
    
    try:
        # 연산을 위해 Decimal로 변환
        decimal_value = self._ensure_decimal(value)
        
        # 원본 타입에 따라 적절한 타입으로 변환하여 반환
        if original_type is int:
            return int(decimal_value)
        elif original_type is float:
            return float(decimal_value)
        elif isinstance(value, str):
            try:
                return float(decimal_value)
            except (ValueError, TypeError, ArithmeticError):
                return 0
        else:
            return float(decimal_value)
            
    except Exception as e:
        _LOGGER.warning(f"Failed to process numeric field: {value}, error: {e}")
        return 0

def _ensure_decimal(self, value: Any) -> Decimal:
    """값을 Decimal 타입으로 안전 변환"""
    if value is None:
        return Decimal("0")
    
    try:
        if isinstance(value, Decimal):
            return value
        elif isinstance(value, (int, float)):
            return Decimal(str(value))  # 정확성을 위해 문자열로 변환
        elif isinstance(value, str):
            cleaned_value = value.strip()
            if not cleaned_value or cleaned_value.lower() in ["null", "none", "n/a"]:
                return Decimal("0")
            return Decimal(cleaned_value)
        else:
            return Decimal("0")
    except Exception as e:
        _LOGGER.warning(f"Failed to convert to Decimal: {value}, error: {e}")
        return Decimal("0")
```

### 2. JSON 직렬화 안전성

```python
def _sanitize_for_serialization(self, data: dict) -> dict:
    """모든 데이터를 JSON 직렬화 가능한 타입으로 변환"""
    
    import pandas as pd
    import numpy as np
    from datetime import datetime, date
    from decimal import Decimal
    
    def convert_value(value):
        # None 체크 우선
        if value is None:
            return None
        
        # Pandas NA 안전 체크
        try:
            if pd.isna(value):
                return None
        except (TypeError, ValueError):
            pass
        
        # 타입별 변환
        if isinstance(value, (pd.Timestamp, pd.Timedelta)):
            return str(value)
        elif isinstance(value, (np.integer, np.floating)):
            return value.item()
        elif isinstance(value, np.ndarray):
            return [convert_value(item) for item in value]
        elif isinstance(value, (datetime, date)):
            return value.isoformat()
        elif isinstance(value, Decimal):
            return float(value)
        elif isinstance(value, dict):
            return {k: convert_value(v) for k, v in value.items()}
        elif isinstance(value, (list, tuple)):
            return [convert_value(item) for item in value]
        else:
            return value
    
    # 전체 데이터 변환
    sanitized_data = {}
    for key, value in data.items():
        try:
            sanitized_value = convert_value(value)
            
            # SpaceONE data 필드 특별 처리
            if key == "data" and not isinstance(sanitized_value, dict):
                sanitized_data[key] = {}
            else:
                sanitized_data[key] = sanitized_value
                
        except Exception as e:
            _LOGGER.warning(f"Failed to sanitize field {key}: {e}")
            if key == "data":
                sanitized_data[key] = {}  # data 필드는 빈 딕셔너리로 보장
            else:
                sanitized_data[key] = str(value)
    
    return sanitized_data
```

### 1. 컴파일된 매핑 규칙

```python
def _compile_mappings(self):
    """매핑 설정을 컴파일하여 실행 성능 최적화"""
    
    default_mapping = self._get_default_mapping(self.provider)
    final_mapping = {**default_mapping, **self.mapping_config}
    
    for field_name, mapping_rule in final_mapping.items():
        self.compiled_mappings[field_name] = self._compile_single_mapping(mapping_rule)

def _compile_single_mapping(self, mapping_rule) -> Callable:
    """단일 매핑 규칙을 실행 가능한 함수로 컴파일"""
    
    if isinstance(mapping_rule, str):
        # 단순 필드 매핑
        return lambda data, rule=mapping_rule: self._get_nested_value(data, rule)
    
    elif isinstance(mapping_rule, dict):
        if "field" in mapping_rule:
            # 필드 + 변환 + fallback 매핑
            return self._compile_field_mapping(mapping_rule)
        elif "expression" in mapping_rule:
            # 표현식 매핑
            return self._compile_expression_mapping(mapping_rule)
        elif "constant" in mapping_rule:
            # 상수 매핑
            return self._compile_constant_mapping(mapping_rule)
        else:
            # 복합 딕셔너리 매핑
            return self._compile_complex_dict_mapping(mapping_rule)
    
    elif callable(mapping_rule):
        # 함수 매핑
        return mapping_rule
    
    # 기본값 반환
    return lambda data: mapping_rule
```

def _track_daily_count(self, billed_date: str):
    """일별 레코드 카운트 추적 및 주기적 로깅"""
    
    if not billed_date or billed_date == "unknown":
        billed_date = "unknown_date"
    
    # 카운트 증가
    self.daily_count_tracker[billed_date] = (
        self.daily_count_tracker.get(billed_date, 0) + 1
    )
    self.total_processed_count += 1
    
    # 주기적 요약 로깅 (성능 고려)
    if self.total_processed_count % 100 == 0:
        self._log_daily_count_summary(is_intermediate=True)
    
    if self.total_processed_count % 1000 == 0:
        self._log_daily_count_summary(is_intermediate=False)

def log_final_daily_count_summary(self):
    """최종 일별 카운트 요약 로깅"""
    
    _LOGGER.info("[FieldMapper] 🎯 === 최종 일별 카운트 요약 ===")
    _LOGGER.info(f"[FieldMapper] 📊 총 처리 레코드: {self.total_processed_count:,}개")
    
    if self.daily_count_tracker:
        # 처리 기간 정보
        min_date = min(self.daily_count_tracker.keys())
        max_date = max(self.daily_count_tracker.keys())
        _LOGGER.info(f"[FieldMapper] 📅 처리 기간: {min_date} ~ {max_date}")
        _LOGGER.info(f"[FieldMapper] 🗓️ 고유 날짜 수: {len(self.daily_count_tracker)}개")
        
        # 날짜별 상세 정보
        sorted_dates = sorted(self.daily_count_tracker.items())
        for date, count in sorted_dates:
            percentage = (count / self.total_processed_count) * 100
            _LOGGER.info(f"[FieldMapper]   📅 {date}: {count:,}개 레코드 ({percentage:.2f}%)")
        
        # 최다/최소 처리 날짜
        max_date_info = max(self.daily_count_tracker.items(), key=lambda x: x[1])
        min_date_info = min(self.daily_count_tracker.items(), key=lambda x: x[1])

def validate_mapping_config(self, mapping_config: dict) -> list:
    """매핑 설정 유효성 검증"""
    
    issues = []
    required_fields = ["cost", "billed_date", "provider"]
    
    for field in required_fields:
        if field not in mapping_config:
            issues.append(f"Missing required mapping for field: {field}")
    
    # 중첩 경로 문법 검증
    for field_name, mapping_rule in mapping_config.items():
        if isinstance(mapping_rule, str) and "." in mapping_rule:
            # 중첩 경로 검증
            if not re.match(r'^[a-zA-Z_][a-zA-Z0-9_]*(\.[a-zA-Z_][a-zA-Z0-9_]*)*$', mapping_rule):
                issues.append(f"Invalid nested path syntax: {mapping_rule}")
        
        elif isinstance(mapping_rule, dict):
            if "field" in mapping_rule:
                # 필드 매핑 검증
                if not isinstance(mapping_rule["field"], str):
                    issues.append(f"Invalid field mapping for {field_name}: field must be string")
                
                # 변환 함수 검증
                if "transform" in mapping_rule:
                    valid_transforms = ["upper", "lower", "decimal", "date_format", "json_parse"]
                    if mapping_rule["transform"] not in valid_transforms:
                        issues.append(f"Unknown transform: {mapping_rule['transform']}")
    
    return issues
